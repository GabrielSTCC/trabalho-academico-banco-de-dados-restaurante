1. Diagrama Conceitual

erDiagram
    CLIENTE ||--o{ PEDIDO : realiza
    MESA ||--o{ PEDIDO : recebe
    PEDIDO ||--o{ ITEM_PEDIDO : contem
    PRATO ||--o{ ITEM_PEDIDO : compoe

    CLIENTE {
        int id_cliente PK
        string nome
        string telefone
        string email
        date data_cadastro
    }

    MESA {
        int id_mesa PK
        int numero
        int capacidade
        string status
    }

    PRATO {
        int id_prato PK
        string nome
        string categoria
        decimal preco
    }

    PEDIDO {
        int id_pedido PK
        datetime data_pedido
        string status
    }

    ITEM_PEDIDO {
        int id_item PK
        int quantidade
        decimal subtotal
    }

2. Diagrama Lógico

erDiagram
    Cliente ||--o{ Pedido : "id_cliente FK"
    Mesa ||--o{ Pedido : "id_mesa FK"
    Pedido ||--o{ Item_Pedido : "id_pedido FK"
    Prato ||--o{ Item_Pedido : "id_prato FK"

    Cliente {
        INT id_cliente PK
        VARCHAR nome
        VARCHAR telefone
        VARCHAR email
        DATE data_cadastro
    }

    Mesa {
        INT id_mesa PK
        INT numero
        INT capacidade
        ENUM status
    }

    Prato {
        INT id_prato PK
        VARCHAR nome
        VARCHAR categoria
        DECIMAL preco
    }

    Pedido {
        INT id_pedido PK
        DATETIME data_pedido
        INT id_cliente FK
        INT id_mesa FK
        ENUM status
    }

    Item_Pedido {
        INT id_item PK
        INT id_pedido FK
        INT id_prato FK
        INT quantidade
        DECIMAL subtotal
    }
